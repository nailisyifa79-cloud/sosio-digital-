            hasil: []
        };

        function loadData() {
            try {
                const saved = localStorage.getItem('sosioDigitalData');
                if (saved) {
                    const parsed = JSON.parse(saved);
                    dataStore.siswa = parsed.siswa || dataStore.siswa;
                    dataStore.hasil = parsed.hasil || [];
                }
            } catch (e) {}

        }

        function renderDropdowns() {
            const selects = [
                'fSuka1', 'fSuka2', 'fKurang1', 'fKurang2', 'fPemimpin', 'fKolab',
                'mSuka1', 'mSuka2', 'mKurang1', 'mKurang2', 'mPemimpin', 'mKolab'

            selects.forEach(id => {
                const sel = document.getElementById(id);
                if (!sel) return;
                const current = sel.value;
                sel.innerHTML = '<option value="">▼ Pilih teman</option>';
                dataStore.siswa.forEach(nama => {
                    const opt = document.createElement('option');
                    opt.value = nama;
                    opt.textContent = nama;
                    sel.appendChild(opt);
                });
                if (current && dataStore.siswa.includes(current)) {

                }

        }

        function renderSiswaList() {
            const container = document.getElementById('siswaList');
            if (!container) return;

        }

        function renderDashboard() {
            const hasil = dataStore.hasil;
            const total = hasil.length;
            document.getElementById('totalSiswa').textContent = total;

            const stats = {};
            dataStore.siswa.forEach(n => {
                stats[n] = { suka: 0, kurang: 0, pemimpin: 0, kolab: 0, total: 0 };
            });

            hasil.forEach(h => {
                const fields = ['suka1', 'suka2', 'kurang1', 'kurang2', 'pemimpin', 'kolab'];
                const map = {
                    suka1: 'suka',
                    suka2: 'suka',
                    kurang1: 'kurang',
                    kurang2: 'kurang',
                    pemimpin: 'pemimpin',
                    kolab: 'kolab'
                };
                fields.forEach(f => {
                    const val = h[f];
                    if (val && stats[val]) {
                        stats[val][map[f]] = (stats[val][map[f]] || 0) + 1;
                        stats[val].total = (stats[val].total || 0) + 1;
                    }
                });
            });

            let populer = '-',
                popJum = 0;
            let pemimpin = '-',
                pemJum = 0;
            let kolab = '-',
                kolJum = 0;

            Object.keys(stats).forEach(n => {
                if (stats[n].suka > popJum) { popJum = stats[n].suka;
                    populer = n; }
                if (stats[n].pemimpin > pemJum) { pemJum = stats[n].pemimpin;
                    pemimpin = n; }
                if (stats[n].kolab > kolJum) { kolJum = stats[n].kolab;
                    kolab = n; }
            });

            document.getElementById('populerSiswa').textContent = populer;
            document.getElementById('populerJumlah').textContent = popJum + ' suka';
            document.getElementById('pemimpinSiswa').textContent = pemimpin;
            document.getElementById('pemimpinJumlah').textContent = pemJum + ' pilihan';
            document.getElementById('kolabSiswa').textContent = kolab;
            document.getElementById('kolabJumlah').textContent = kolJum + ' pilihan';

            renderChart(stats);
            renderTable(stats);
            renderDataTable();
        }

        function renderChart(stats) {
            const container = document.getElementById('chartContainer');
            const names = dataStore.siswa.slice(0, 12);
            if (names.length === 0 || dataStore.hasil.length === 0) {
                container.innerHTML =
                    `<div style="text-align:center; color:#A0AEC0; padding:20px; font-size:13px;">Belum ada data. Silakan isi kuesioner atau entri manual.</div>`;
                return;
            }
            const maxVal = Math.max(1, ...names.map(n => stats[n]?.suka || 0));
            let html = '';
            names.forEach(n => {
                const s = stats[n] || { suka: 0, kurang: 0, pemimpin: 0, kolab: 0 };
                const total = s.suka + s.kurang + s.pemimpin + s.kolab;
                if (total === 0) return;
                const pct = maxVal > 0 ? (s.suka / maxVal) * 100 : 0;
                html += `
                        <div class="chart-bar">
                            <span class="clabel">${n}</span>
                            <div class="track">
                                <div class="fill blue" style="width:${Math.min(pct, 100)}%;"></div>
                            </div>
                            <span class="cvalue">${s.suka}</span>
                        </div>
                    `;
            });
            if (!html) {
                html =
                    `<div style="text-align:center; color:#A0AEC0; padding:20px; font-size:13px;">Belum ada data yang cukup untuk grafik.</div>`;
            }
            container.innerHTML = html;
        }

        function renderTable(stats) {
            const tbody = document.getElementById('tableBody');
            const names = dataStore.siswa;
            if (names.length === 0 || dataStore.hasil.length === 0) {
                tbody.innerHTML =
                    `<tr><td colspan="7" style="text-align:center; color:#A0AEC0; padding:20px;">Belum ada data siswa.</td></tr>`;
                return;
            }
            let html = '';
            let no = 1;
            const sorted = names.map(n => {
                const s = stats[n] || { suka: 0, kurang: 0, pemimpin: 0, kolab: 0 };
                return { nama: n, ...s };
            }).sort((a, b) => (b.suka + b.pemimpin + b.kolab) - (a.suka + a.pemimpin + a.kolab));

            sorted.forEach(s => {
                const total = s.suka + s.kurang + s.pemimpin + s.kolab;
                let badge = 'silver';
                if (no === 1) badge = 'gold';
                else if (no === 2 || no === 3) badge = 'bronze';
                html += `
                        <tr>
                            <td>${no}</td>
                            <td><strong>${s.nama}</strong></td>
                            <td>${s.suka}</td>
                            <td>${s.kurang}</td>
                            <td>${s.pemimpin}</td>
                            <td>${s.kolab}</td>
                            <td><span class="badge-poin ${badge}">${total}</span></td>
                        </tr>
                    `;
                no++;
            });
            tbody.innerHTML = html;
        }

        // ============================================================
        // SISWA MANAGEMENT
        // ============================================================
        function tambahSiswa() {
            const input = document.getElementById('inputNamaSiswa');
            const nama = input.value.trim();
            if (!nama) {
                showToast('Masukkan nama siswa!', 'error');
                return;
            }
            if (dataStore.siswa.includes(nama)) {
                showToast('Nama sudah ada!', 'error');
                return;
            }
            dataStore.siswa.push(nama);
            input.value = '';
            saveData();
            showToast('✅ ' + nama + ' berhasil ditambahkan!', 'success');
        }

        function tambahSiswaBulk() {
            const input = prompt('Masukkan nama siswa (pisahkan dengan koma):\nContoh: Andi, Budi, Citra');
            if (!input) return;
            const names = input.split(',').map(s => s.trim()).filter(s => s);
            let added = 0;
            names.forEach(n => {
                if (!dataStore.siswa.includes(n)) {
                    dataStore.siswa.push(n);
                    added++;
                }
            });
            saveData();
            showToast('✅ ' + added + ' siswa berhasil ditambahkan!', 'success');
        }

        function hapusSiswa(index) {
            if (confirm('Hapus ' + dataStore.siswa[index] + '?')) {
                dataStore.siswa.splice(index, 1);
                saveData();
                showToast('🗑️ Siswa dihapus', 'info');
            }
        }

        function resetSiswa() {
            if (confirm('Hapus semua siswa? Data hasil akan tetap tersimpan.')) {
                dataStore.siswa = [];
                saveData();
                showToast('🗑️ Semua siswa dihapus', 'info');
            }
        }

        // FORM SUBMIT
        // ============================================================
        function submitForm() {
            const nama = document.getElementById('fNama').value.trim();
            const kelas = document.getElementById('fKelas').value;
            const suka1 = document.getElementById('fSuka1').value;
            const suka2 = document.getElementById('fSuka2').value;
            const kurang1 = document.getElementById('fKurang1').value;
            const kurang2 = document.getElementById('fKurang2').value;
            const pemimpin = document.getElementById('fPemimpin').value;
            const kolab = document.getElementById('fKolab').value;

            if (!nama || !kelas || !suka1 || !suka2 || !kurang1 || !kurang2 || !pemimpin || !kolab) {
                showToast('⚠️ Semua data harus diisi!', 'error');
                return;
            }
            if (suka1 === suka2) {
                showToast('⚠️ Teman disukai 1 dan 2 harus berbeda!', 'error');
                return;
            }
            if (kurang1 === kurang2) {
                showToast('⚠️ Teman kurang nyaman 1 dan 2 harus berbeda!', 'error');
                return;
            }

            dataStore.hasil.push({
                nama,
                kelas,
                suka1,
                suka2,
                kurang1,
                kurang2,
                pemimpin,
                kolab,
                timestamp: new Date().toISOString()
            });

            saveData();
            document.getElementById('fNama').value = '';
            document.getElementById('fKelas').value = '';
            ['fSuka1', 'fSuka2', 'fKurang1', 'fKurang2', 'fPemimpin', 'fKolab'].forEach(id => {
                document.getElementById(id).value = '';
            });

            showToast('✅ Jawaban berhasil disimpan!', 'success');
            showPage('page-success');
        }

        // ============================================================
        // PAGE NAVIGATION
        // ============================================================
        function showPage(pageId) {
            const pages = ['page-home', 'page-form', 'page-success', 'page-dashboard', 'page-manual-input'];
            pages.forEach(id => {
                document.getElementById(id).classList.toggle('hidden', id !== pageId);
            });
            document.querySelector('.phone').scrollTop = 0;
            if (pageId === 'page-dashboard' || pageId === 'page-home') {
                renderSemua();
            }
            if (pageId === 'page-form') {
                updateProgress();
            }
            if (pageId === 'page-manual-input') {
                renderDropdowns();
            }
        }

        // ============================================================
        // TAB SWITCH
        // ============================================================
        function switchTab(btn, tabId) {
            document.querySelectorAll('.tab-btn').forEach(b => b.classList.remove('active'));
            btn.classList.add('active');
            ['tabDashboard', 'tabSiswa', 'tabData'].forEach(id => {
                document.getElementById(id).classList.toggle('hidden', id !== tabId);
            });
            if (tabId === 'tabSiswa') renderSiswaList();
            if (tabId === 'tabData') renderDataTable();
        }

        // PROGRESS
        // ============================================================
        function updateProgress() {
            const fields = [
                document.getElementById('fNama'),
                document.getElementById('fKelas'),
                document.getElementById('fSuka1'),
                document.getElementById('fSuka2'),
                document.getElementById('fKurang1'),
                document.getElementById('fKurang2'),
                document.getElementById('fPemimpin'),
                document.getElementById('fKolab')
            ];
            let filled = 0;
            let total = fields.length;
            fields.forEach(f => {
                if (f && f.value && f.value !== '') filled++;
            });
            const pct = total > 0 ? Math.round((filled / total) * 100) : 0;
            const fill = document.getElementById('progressFill');
            const text = document.getElementById('progressText');
            if (fill) fill.style.width = Math.min(pct, 100) + '%';
            if (text) text.textContent = Math.min(pct, 100) + '% selesai';
        }

            let csv = 'No,Nama,Kelas,Teman Disukai 1,Teman Disukai 2,Kurang Nyaman 1,Kurang Nyaman 2,Jiwa Pemimpin,Mudah Kolaborasi\n';
            dataStore.hasil.forEach((h, i) => {
                csv +=
                    `${i+1},${h.nama || ''},${h.kelas || ''},${h.suka1 || ''},${h.suka2 || ''},${h.kurang1 || ''},${h.kurang2 || ''},${h.pemimpin || ''},${h.kolab || ''}\n`;
            });
            const blob = new Blob([csv], { type: 'text/csv;charset=utf-8;' });
            const link = document.createElement('a');
            link.href = URL.createObjectURL(blob);
            link.download = `Hasil_Sosiogram_${new Date().toISOString().slice(0,10)}.csv`;
            link.click();
            showToast('📥 Data berhasil diekspor!', 'success');
        }
