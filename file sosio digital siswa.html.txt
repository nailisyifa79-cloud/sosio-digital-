    <script>
        // ============================================================
        // DATA STORE (tersimpan di localStorage)
        // ============================================================
        let dataStore = {
            siswa: ['Andi', 'Budi', 'Citra', 'Dedi', 'Eka', 'Fani', 'Gita', 'Hadi', 'Indra', 'Joko'],
            hasil: []
        };

        // Load dari localStorage
        function loadData() {
            try {
                const saved = localStorage.getItem('sosioDigitalData');
                if (saved) {
                    const parsed = JSON.parse(saved);
                    dataStore.siswa = parsed.siswa || dataStore.siswa;
                    dataStore.hasil = parsed.hasil || [];
                }
            } catch (e) {}
            renderDropdowns();
        }

        function saveData() {
            localStorage.setItem('sosioDigitalData', JSON.stringify(dataStore));
        }

        // ============================================================
        // RENDER DROPDOWN
        // ============================================================
        function renderDropdowns() {
            const selects = ['fSuka1', 'fSuka2', 'fKurang1', 'fKurang2', 'fPemimpin', 'fKolab'];
            selects.forEach(id => {
                const sel = document.getElementById(id);
                if (!sel) return;
                const current = sel.value;
                sel.innerHTML = '<option value="">▼ Pilih teman</option>';
                dataStore.siswa.forEach(nama => {
                    const opt = document.createElement('option');
                    opt.value = nama;
                    opt.textContent = nama;
                    sel.appendChild(opt);
                });
                if (current && dataStore.siswa.includes(current)) {
                    sel.value = current;
                }
            });
        }

        // ============================================================
        // PROGRESS BAR
        // ============================================================
        function updateProgress() {
            const fields = [
                document.getElementById('fNama'),
                document.getElementById('fKelas'),
                document.getElementById('fSuka1'),
                document.getElementById('fSuka2'),
                document.getElementById('fKurang1'),
                document.getElementById('fKurang2'),
                document.getElementById('fPemimpin'),
                document.getElementById('fKolab')
            ];

            let filled = 0;
            let total = fields.length;

            fields.forEach(f => {
                if (f && f.value && f.value !== '') filled++;
            });

            const pct = total > 0 ? Math.round((filled / total) * 100) : 0;

            const fill = document.getElementById('progressFill');
            const text = document.getElementById('progressText');

            if (fill) fill.style.width = Math.min(pct, 100) + '%';
            if (text) text.textContent = Math.min(pct, 100) + '% selesai';
        }

        // ============================================================
        // SUBMIT FORM
        // ============================================================
        function submitForm() {
            const nama = document.getElementById('fNama').value.trim();
            const kelas = document.getElementById('fKelas').value;
            const suka1 = document.getElementById('fSuka1').value;
            const suka2 = document.getElementById('fSuka2').value;
            const kurang1 = document.getElementById('fKurang1').value;
            const kurang2 = document.getElementById('fKurang2').value;
            const pemimpin = document.getElementById('fPemimpin').value;
            const kolab = document.getElementById('fKolab').value;

            // Validasi
            if (!nama || !kelas || !suka1 || !suka2 || !kurang1 || !kurang2 || !pemimpin || !kolab) {
                showToast('⚠️ Semua data harus diisi!', 'error');
                return;
            }

            if (suka1 === suka2) {
                showToast('⚠️ Teman disukai 1 dan 2 harus berbeda!', 'error');
                return;
            }

            if (kurang1 === kurang2) {
                showToast('⚠️ Teman kurang nyaman 1 dan 2 harus berbeda!', 'error');
                return;
            }

            // Simpan data
            dataStore.hasil.push({
                nama: nama,
                kelas: kelas,
                suka1: suka1,
                suka2: suka2,
                kurang1: kurang1,
                kurang2: kurang2,
                pemimpin: pemimpin,
                kolab: kolab,
                timestamp: new Date().toISOString()
            });

            saveData();

            // Tampilkan halaman sukses
            document.getElementById('page-form').classList.add('hidden');
            document.getElementById('page-success').classList.remove('hidden');

            showToast('✅ Jawaban berhasil disimpan!', 'success');
        }

        // ============================================================
        // RESET FORM
        // ============================================================
        function resetForm() {
            document.getElementById('fNama').value = '';
            document.getElementById('fKelas').value = '';
            ['fSuka1', 'fSuka2', 'fKurang1', 'fKurang2', 'fPemimpin', 'fKolab'].forEach(id => {
                document.getElementById(id).value = '';
            });

            document.getElementById('page-success').classList.add('hidden');
            document.getElementById('page-form').classList.remove('hidden');

            updateProgress();
            window.scrollTo(0, 0);
        }

        // ============================================================
        // TOAST
        // ============================================================
        function showToast(message, type) {
            const toast = document.getElementById('toast');
            toast.textContent = message;
            toast.className = 'toast ' + (type || 'info');
            setTimeout(() => toast.classList.add('show'), 10);
            clearTimeout(toast._timer);
            toast._timer = setTimeout(() => toast.classList.remove('show'), 3000);
        }

        // ============================================================
        // INIT
        // ============================================================
        loadData();
        updateProgress();

        console.log('🌈 SOSIO DIGITAL - by NaFaS BK');
        console.log('📱 Halaman SISWA - Sosiogram Digital');
        console.log('👥 Jumlah siswa:', dataStore.siswa.length);
    </script>
